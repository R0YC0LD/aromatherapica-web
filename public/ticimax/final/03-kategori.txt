<script>
(function () {
  "use strict";
  function start() {
    var ar = window.AromatherapicaTicimax;
    if (!ar) return window.setTimeout(start, 80);
    ar.addPageClass("category");
    var root = ar.qs(".ProductListContent, .ProductList, .productList, #divIcerik") || document.body;
    root.classList.add("ar-native-catalog");
    ar.enhanceProductCards(root);
    ar.observeProducts(root);
    ar.qsa(".filter, .Filter, .sorting, .sort, [class*='Filtre']", root).forEach(function (node) {
      node.classList.add("ar-native-catalog-control");
    });
  }
  if (document.readyState === "loading") document.addEventListener("DOMContentLoaded", start, { once: true });
  else start();
})();
</script>
