  }

  function imageSource(img) {
    if (!img) return "";
    var source = img.currentSrc || img.getAttribute("src") || img.getAttribute("data-src") || img.getAttribute("data-original") || "";
    if (!source || /noimage|no-image|resim-hazirlaniyor|placeholder|default-product|data:image\/gif/i.test(source)) return "";
    return source;
  }

  function productImage(item) {
    return imageSource(qs(".productImage img, [class*='productImage'] img, img", item));
  }

  function categoryOf(item) {
    var detail = qs(".productDetail", item);
    var value = detail && detail.dataset ? (detail.dataset.category1 || detail.dataset.category || "") : "";
    return String(value || item.getAttribute("data-category") || item.textContent || "").toLocaleLowerCase("tr-TR");
  }

  function assetVisual(source, className, alt) {
    if (source) {
      return '<div class="' + className + '"><img src="' + source + '" alt="' + (alt || "") + '" width="640" height="720" decoding="async"></div>';
    }
    var emblem = (window.AR_THEME_CONFIG && window.AR_THEME_CONFIG.emblemUrl) || "/Uploads/EditorUploads/aromatherapica-emblem.png";
    return '<div class="ar-hero-brand-visual"><img src="' + emblem + '" alt="Aromatherapica amblemi" width="320" height="320"></div>';
  }

    var seen = [];
    return qsa(".productItem, [class~='productItem']").filter(function (item) {
      if (item.closest("#ar-site-header, #ar-site-footer, .ar-home")) return false;
      if (seen.indexOf(item) >= 0) return false;
      seen.push(item);
      return true;
    }).slice(0, 8);
  }

  function sourceRoot(products) {
    if (!products.length) return null;
    return products[0].closest("#divIcerik, .mainContainer, .contentWrapper, .HomePage, main");
  }

  function build(products) {
    if (qs("#ar-main")) return;
    var firstImage = products.map(productImage).find(Boolean) || "";
    var secondImage = products.slice(1).map(productImage).find(Boolean) || firstImage;
    var main = doc.createElement("main");
    main.id = "ar-main";
    main.className = "ar-home";
    main.innerHTML =
      '<section class="ar-hero">' +
        '<div class="ar-hero-media">' + assetVisual(firstImage, "ar-hero-product", "") + "</div>" +
        '<div class="ar-hero-copy">' +
          '<p class="ar-eyebrow">Doğanın bilgisinden modern bakım ritüellerine</p>' +

      '<section class="ar-product-zone" aria-labelledby="ar-products-title">' +
        '<div class="ar-section-head ar-reveal"><div><p class="ar-eyebrow">Aromatherapica seçkisi</p><h2 id="ar-products-title">Çok sevilenler</h2></div>' +
          '<div class="ar-filters" role="group" aria-label="Ürün filtreleri"><button class="is-active" type="button" data-ar-filter="all">Tümü</button><button type="button" data-ar-filter="cilt">Cilt Bakımı</button><button type="button" data-ar-filter="yağ">Uçucu Yağlar</button><button type="button" data-ar-filter="özel">Özel Bakım</button></div>' +
        "</div>" +
        '<div class="ar-product-grid" id="ar-real-products"></div>' +
      "</section>" +
      '<section class="ar-rituals">' +

        "</div>" +

  }

  function bodyInsert(node, before) {
    if (before && before.parentNode) before.parentNode.insertBefore(node, before);
    else doc.body.appendChild(node);
  }

  function bindFilters(grid) {
    qsa("[data-ar-filter]").forEach(function (button) {
      button.addEventListener("click", function () {
        var filter = button.getAttribute("data-ar-filter");
        qsa("[data-ar-filter]").forEach(function (item) { item.classList.toggle("is-active", item === button); });
        // Ticimax native .productItem nodes (enhanced as .ar-product-card)
        qsa(".ar-product-card, .productItem", grid).forEach(function (card) {
          var category = card.dataset.arHomeCategory || categoryOf(card);
          card.hidden = filter !== "all" && category.indexOf(filter) < 0;
        });
      });
    });
  }

  function watchLateProducts(grid) {
    if (!window.MutationObserver || !grid) return;
    var tries = 0;
    var timer = window.setInterval(function () {
      tries += 1;
      var extras = collectNativeProducts().filter(function (item) {
        return !grid.contains(item);
      });
      extras.forEach(function (item) {
        item.dataset.arHomeCategory = categoryOf(item);
        grid.appendChild(item);
      });
      if (extras.length && qs(".ar-empty-state", grid)) {
        var empty = qs(".ar-empty-state", grid);
        if (empty && empty.parentNode) empty.parentNode.removeChild(empty);
      }
      if (tries >= 20) window.clearInterval(timer);
    }, 500);
  }

  function reveal(scope) {
    var items = qsa(".ar-reveal", scope);
    if (!("IntersectionObserver" in window)) {
      items.forEach(function (item) { item.classList.add("is-visible"); });
      return;
    }
    var observer = new IntersectionObserver(function (entries) {
      entries.forEach(function (entry) {
        if (!entry.isIntersecting) return;
        entry.target.classList.add("is-visible");
        observer.unobserve(entry.target);
      });
    }, { threshold: .07, rootMargin: "0px 0px -6% 0px" });
    items.forEach(function (item) { observer.observe(item); });
  }

  function init() {
    if (!doc.body || !doc.body.classList.contains("ar-page-home")) return;
    var products = collectNativeProducts();
    var source = sourceRoot(products);
    build(products);
    if (source && source !== doc.body && !source.contains(qs("#ar-main"))) {
      source.classList.add("ar-native-home-source");
    }
  }

  if (doc.readyState === "loading") doc.addEventListener("DOMContentLoaded", init, { once: true });
  else init();
  window.addEventListener("ar:core-ready", init, { once: true });
})();
