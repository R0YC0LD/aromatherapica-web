<script>
(function () {
  "use strict";
  function start() {
    var ar = window.AromatherapicaTicimax;
    if (!ar) return window.setTimeout(start, 80);
    ar.addPageClass("member-success");
    var main = ar.qs("main, #mainHolder, .mainContainer, #divIcerik");
    if (main) {
      main.setAttribute("role", "status");
      main.setAttribute("aria-live", "polite");
    }
  }
  if (document.readyState === "loading") document.addEventListener("DOMContentLoaded", start, { once: true });
  else start();
})();
</script>
