    var drawer = qs("#ar-mobile-drawer");
    var search = qs("#ar-search-panel");
    var scrim = qs(".ar-scrim");
    var target = name === "menu" ? drawer : search;
    if (!target) return;

    [drawer, search].forEach(function (node) {
      if (!node) return;
      var active = node === target && open;
      node.classList.toggle(node === drawer ? "is-open" : "is-open", active);
      node.setAttribute("aria-hidden", active ? "false" : "true");
    });
    qsa(".ar-menu-trigger").forEach(function (button) { button.setAttribute("aria-expanded", name === "menu" && open ? "true" : "false"); });
    qsa(".ar-search-trigger").forEach(function (button) { button.setAttribute("aria-expanded", name === "search" && open ? "true" : "false"); });
    scrim.classList.toggle("is-visible", open);
    root.classList.toggle("ar-layer-open", open);
    if (open) {
      var input = qs("input", target);
      if (input) window.setTimeout(function () { input.focus(); }, 160);
    }
  }

  function bindHeader() {
    qsa(".ar-menu-trigger").forEach(function (button) {
      button.addEventListener("click", function () { setLayer("menu", true); });
    });
    qsa(".ar-search-trigger").forEach(function (button) {
      button.addEventListener("click", function () { setLayer("search", true); });
    });
    qsa(".ar-close").forEach(function (button) {
      button.addEventListener("click", function () { setLayer("menu", false); });
    });
    qs(".ar-scrim").addEventListener("click", function () { setLayer("menu", false); });
    doc.addEventListener("keydown", function (event) {
      if (event.key === "Escape") setLayer("menu", false);
    });
    qs(".ar-back-top").addEventListener("click", function () {
      window.scrollTo({ top: 0, behavior: "smooth" });
    });
    window.addEventListener("scroll", scheduleScroll, { passive: true });
  }

  function scheduleScroll() {
    if (raf) return;
    raf = window.requestAnimationFrame(function () {
      raf = 0;
      var current = Math.max(0, window.scrollY || 0);
      var goingDown = current > lastScroll + 4;
      var goingUp = current < lastScroll - 4;
      var announcement = qs(".ar-announcement");
      var header = qs(".ar-header");
      var topButton = qs(".ar-back-top");
      var progress = qs(".ar-progress > span");

      if (announcement) {
        if (goingDown && current > 90) announcement.classList.add("is-hidden");
        if (goingUp || current < 24) announcement.classList.remove("is-hidden");
      }
      if (header) header.classList.toggle("is-scrolled", current > 20);
      if (topButton) topButton.classList.toggle("is-visible", current > 520);
      if (progress) {
        var max = Math.max(1, doc.documentElement.scrollHeight - window.innerHeight);
        progress.style.transform = "scaleX(" + Math.min(1, current / max) + ")";
      }
      lastScroll = current;
    });
  }

  function updateCartCount() {
    var source = qs(
      ".CartProductNumber, .cart-product-count, .sepetUrunSayisi, [class*='SepetUrunSayisi'], [data-cart-count]"
    );
    var match = text(source).match(/\d+/);
    qsa(".ar-cart-count").forEach(function (node) { node.textContent = match ? match[0] : "0"; });
  }

  function ensureImage(item) {
    var image = qs(".productImage img, [class*='productImage'] img, img", item);
    if (!image) return;
    var source = realImageSource(image);
    if (source && !image.getAttribute("src")) image.setAttribute("src", source);
    image.setAttribute("loading", "lazy");
    image.setAttribute("decoding", "async");
    if (!image.getAttribute("width")) image.setAttribute("width", "520");
    if (!image.getAttribute("height")) image.setAttribute("height", "600");
    image.addEventListener("error", function () {
      image.classList.add("ar-image-failed");
      var host = image.parentElement;
      if (host && !qs(".ar-no-image", host)) {
        var empty = doc.createElement("span");
        empty.className = "ar-no-image";
        empty.textContent = "Bu ürüne ait görsel henüz eklenmemiş.";
        host.appendChild(empty);
      }
    }, { once: true });
  }

  function enhanceProduct(item) {
    if (!item || item.dataset.arEnhanced === "true") return;
    item.dataset.arEnhanced = "true";
    item.classList.add("ar-product-card");
    ensureImage(item);

    var detail = qs(".productDetail", item) || item;
    var add = qs(".btnAddToCart, .AddToCart, .addToCart, [class*='addToCart']", item);
    var favorite = qs(".favoriteslist, .favorite, .addToFavorite, [class*='favorite' i]", item);
    var nameLink = qs(".productName a, [class*='productName'] a, .productImage a", item);

    if (add && !add.getAttribute("aria-label")) add.setAttribute("aria-label", "Sepete ekle");
    if (favorite && !favorite.getAttribute("aria-label")) favorite.setAttribute("aria-label", "Favorilere ekle");

    var actions = qs(".ar-native-actions", item);
    if (!actions) {
      actions = doc.createElement("div");
      actions.className = "ar-native-actions";
      detail.appendChild(actions);
    }
    if (add && add.parentElement !== actions) actions.appendChild(add);
    if (!add && nameLink && !qs(".ar-detail-link", actions)) {
      var detailLink = doc.createElement("a");
      detailLink.className = "ar-detail-link";
      detailLink.href = safeUrl(nameLink.getAttribute("href"), "/urunler");
      detailLink.textContent = "Ürünü incele";
      actions.appendChild(detailLink);
    }
  }

  function enhanceProducts(scope) {
    qsa(".productItem, [class~='productItem']", scope || doc).forEach(enhanceProduct);
  }

  function editorInstance(wrapper) {
    if (!wrapper) return null;
    if (wrapper.CodeMirror && typeof wrapper.CodeMirror.refresh === "function") {
      return function () { wrapper.CodeMirror.refresh(); };
    }
    if (wrapper.env && wrapper.env.editor && typeof wrapper.env.editor.resize === "function") {
      return function () { wrapper.env.editor.resize(true); };
    }
    if (window.ace && wrapper.id) {
      try {
        var aceEditor = window.ace.edit(wrapper);
        if (aceEditor) return function () { aceEditor.resize(true); };
      } catch (error) {}
    }
    if (window.monaco && window.monaco.editor && typeof window.monaco.editor.getEditors === "function") {
      return function () {
        window.monaco.editor.getEditors().forEach(function (editor) { editor.layout(); });
      };
    }
    return null;
  }

  function refreshEditors(scope) {
    qsa(".CodeMirror, .ace_editor, .monaco-editor", scope || doc).forEach(function (wrapper) {
      if (wrapper.offsetWidth === 0 || wrapper.offsetHeight === 0) return;
      var refresh = editorInstance(wrapper);
      if (refresh) window.requestAnimationFrame(refresh);
    });
  }

  function watchDynamicContent() {
    if (!window.MutationObserver) return;
    var timer = 0;
    var observer = new MutationObserver(function (mutations) {
      var shouldEnhance = mutations.some(function (mutation) { return mutation.addedNodes.length > 0; });
      if (!shouldEnhance) return;
      window.clearTimeout(timer);
      timer = window.setTimeout(function () {
        enhanceProducts();
        updateCartCount();
        refreshEditors();
      }, 80);
    });
    observer.observe(body, { childList: true, subtree: true });

    qsa("[role='tab'], .accordion, [data-toggle], button").forEach(function (control) {
      control.addEventListener("click", function () {
        window.setTimeout(function () { refreshEditors(); }, 50);
        window.setTimeout(function () { refreshEditors(); }, 280);
      });
    });
    window.addEventListener("resize", function () { refreshEditors(); }, { passive: true });
  }

  function revealContent() {
    var items = qsa(".ar-reveal");
    if (!("IntersectionObserver" in window)) {
      items.forEach(function (item) { item.classList.add("is-visible"); });
      return;
    }
    var observer = new IntersectionObserver(function (entries) {
      entries.forEach(function (entry) {
        if (!entry.isIntersecting) return;
        entry.target.classList.add("is-visible");
        observer.unobserve(entry.target);
      });
    }, { threshold: .08, rootMargin: "0px 0px -7% 0px" });
    items.forEach(function (item) { observer.observe(item); });
  }

  function markPage() {
    var path = String(window.location.pathname || "").toLocaleLowerCase("tr-TR");
    if (/urun.*detay|product|-[0-9]+$/.test(path)) body.classList.add("ar-page-product");
    else if (/kategori|marka|arama|urunler/.test(path)) body.classList.add("ar-page-list");
    else if (/sepet|checkout|siparis/.test(path)) body.classList.add("ar-page-cart");
    else if (path === "/" || /default|anasayfa/.test(path)) body.classList.add("ar-page-home");
  }

  function init() {
    body = doc.body;
    if (!body) return;
    root.classList.add("ar-v2");
    body.classList.add("ar-v2");
    markPage();
    mountHeader();
    enhanceProducts();
    mountFooter();
    watchDynamicContent();
    refreshEditors();
    revealContent();
    scheduleScroll();
    window.dispatchEvent(new CustomEvent("ar:core-ready"));
  }

  if (doc.readyState === "loading") doc.addEventListener("DOMContentLoaded", init, { once: true });
  else init();
})();

(function () {
  "use strict";

  var doc = document;

  function qs(selector, scope) {
    try { return (scope || doc).querySelector(selector); } catch (error) { return null; }
  }

  function qsa(selector, scope) {
    try { return Array.prototype.slice.call((scope || doc).querySelectorAll(selector)); } catch (error) { return []; }
  }

  function listPage() {
    if (!doc.body.classList.contains("ar-page-list")) return;
    var content = qs("#divIcerik, .mainContainer, main");
    if (!content || content.closest(".ar-home")) return;
    content.classList.add("ar-commerce-page");

    var list = qs(".ProductList, .productList, [class*='ProductList']", content);
    if (list) list.classList.add("ar-product-grid");

    var sidebar = qs(".leftBlock, .categoryFilter, [class*='filter' i]", content);
    if (sidebar) sidebar.classList.add("ar-category-sidebar");

    var heading = qs("h1, .categoryTitle, [class*='categoryTitle']", content);
    if (heading) heading.classList.add("ar-page-title");

    var sort = qs("select[id*='sort' i], select[class*='sort' i], .sortBy", content);
    if (sort && !qs(".ar-category-tools", content)) {
      var tools = doc.createElement("div");
      tools.className = "ar-category-tools";
      var count = qs(".productCount, [class*='productCount']", content);
      if (count) tools.appendChild(count);
      tools.appendChild(sort);
      if (list && list.parentNode) list.parentNode.insertBefore(tools, list);
    }
  }

  function productPage() {
    if (!doc.body.classList.contains("ar-page-product")) return;
    var content = qs("#divIcerik, .mainContainer, main");
    if (!content) return;
    content.classList.add("ar-commerce-page");

    var detail = qs(".ProductDetail, .productDetailContainer, [class*='ProductDetail']", content);
    if (!detail) return;
    detail.classList.add("ar-product-detail");

    qsa(".ProductDetail img, .productDetailContainer img", content).forEach(function (img) {
      img.setAttribute("decoding", "async");
      if (!img.getAttribute("width")) img.setAttribute("width", "760");
      if (!img.getAttribute("height")) img.setAttribute("height", "760");
    });
  }

  function cartAndAccountPages() {
    if (!doc.body.classList.contains("ar-page-cart")) return;
    var content = qs("#divIcerik, .mainContainer, main");
    if (content) content.classList.add("ar-commerce-page");
  }

  function init() {
    if (!doc.body || !doc.body.classList.contains("ar-v2")) return;
    listPage();
    productPage();
    cartAndAccountPages();
  }

  if (doc.readyState === "loading") doc.addEventListener("DOMContentLoaded", init, { once: true });
  else init();
  window.addEventListener("ar:core-ready", init, { once: true });
})();
